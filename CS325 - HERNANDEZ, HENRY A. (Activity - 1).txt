class SimpleRandom:
    """Simple random number generator"""
    _seed = 12345

    @classmethod
    def random(cls):
        cls._seed = (cls._seed * 9301 + 49297) % 233280
        return cls._seed / 233280

    @classmethod
    def randint(cls, a, b):
        return int(a + (cls.random() * (b - a + 1)))

    @classmethod
    def choice(cls, seq):
        return seq[cls.randint(0, len(seq) - 1)] if seq else None


class StarField:
    """Manages background stars"""

    def __init__(self):
        self.stars = []
        for _ in range(150):
            self.stars.append({
                'x': SimpleRandom.randint(0, SCREEN_WIDTH),
                'y': SimpleRandom.randint(0, SCREEN_HEIGHT),
                'size': SimpleRandom.randint(1, 3),
                'speed': SimpleRandom.random() * 3 + 1
            })

    def update(self):
        for star in self.stars:
            star['y'] -= star['speed']
            if star['y'] < 0:
                star['y'] = SCREEN_HEIGHT
                star['x'] = SimpleRandom.randint(0, SCREEN_WIDTH)

    def draw(self, canvas):
        with canvas:
            Color(1, 1, 1, 1)
            for star in self.stars:
                px = int(star['x'] // PIXEL_SIZE) * PIXEL_SIZE
                py = int(star['y'] // PIXEL_SIZE) * PIXEL_SIZE
                size = max(PIXEL_SIZE, int(star['size']) * PIXEL_SIZE)
                Rectangle(pos=(px, py), size=(size, size))


class Particle:
    """Particle effect for explosions"""

    def __init__(self, x, y, color):
        self.x = x
        self.y = y
        self.vx = SimpleRandom.random() * 6 - 3
        self.vy = SimpleRandom.random() * 6 - 3
        self.life = 1.0
        self.color = color
        self.size = SimpleRandom.randint(2, 5)

    def update(self):
        self.x += self.vx
        self.y += self.vy
        self.life -= 0.02
        return self.life > 0

    def draw(self, canvas):
        with canvas:
            Color(self.color[0], self.color[1], self.color[2], self.life)
            size = max(PIXEL_SIZE, int(self.size) * PIXEL_SIZE)
            px = int((self.x - size / 2) // PIXEL_SIZE) * PIXEL_SIZE
            py = int((self.y - size / 2) // PIXEL_SIZE) * PIXEL_SIZE
            Rectangle(pos=(px, py), size=(size, size))

    def setup_keyboard(self):
        """Setup keyboard handling"""
        self._keyboard = Window.request_keyboard(self._keyboard_closed, self)
        self._keyboard.bind(on_key_down=self._on_key_down)
        self._keyboard.bind(on_key_up=self._on_key_up)
        self.keys_pressed = set()

    def _keyboard_closed(self):
        self._keyboard.unbind(on_key_down=self._on_key_down)
        self._keyboard.unbind(on_key_up=self._on_key_up)
        self._keyboard = None

    def _on_key_down(self, keyboard, keycode, text, modifiers):
        self.keys_pressed.add(keycode[1])

        if self.game_state == "menu":
            if keycode[1] == 'spacebar':
                self.start_game()
        elif self.game_state == "game_over":
            if keycode[1] == 'r':
                self.reset_game()
            elif keycode[1] == 'q':
                App.get_running_app().stop()
        elif self.game_state == "victory":
            if keycode[1] == 'r':
                self.reset_game()

    def _on_key_up(self, keyboard, keycode):
        self.keys_pressed.discard(keycode[1])

    def spawn_stage_enemy(self):
        """Spawn one stage-themed regular enemy."""
        if self.game_state != "playing":
            return
        if self.stage_phase != "enemies":
            return

        mode = self.current_profile.get('enemy_mode', 'alien')
        variant_pool = ['basic', 'fast', 'tank']
        variant = SimpleRandom.choice(variant_pool)

        enemy = {
            'x': SimpleRandom.randint(40, SCREEN_WIDTH - 40),
            'y': SCREEN_HEIGHT + SimpleRandom.randint(10, 60),
            'width': 30,
            'height': 30,
            'type': 'enemy',
            'variant': variant,
            'sprite_mode': mode,
            'health': 1,
            'speed': 2 + (self.wave * 0.15),
            'points': 10 + (self.wave * 2),
            'shoot_chance': 0.006,
        }

        if variant == 'fast':
            enemy['width'] = 20
            enemy['height'] = 20
            enemy['speed'] += 1.8
            enemy['points'] += 8
            enemy['shoot_chance'] = 0.003
        elif variant == 'tank':
            enemy['width'] = 40
            enemy['height'] = 40
            enemy['health'] = 3 + (self.wave // 3)
            enemy['speed'] = max(1.0, enemy['speed'] - 1.4)
            enemy['points'] += 16
            enemy['shoot_chance'] = 0.009

        if mode == 'meteor':
            enemy['speed'] += 1.1
            enemy['drift'] = SimpleRandom.random() * 2.0 - 1.0
            enemy['shoot_chance'] = 0.0
        elif mode == 'hybrid':
            enemy['drift'] = (SimpleRandom.random() * 1.8) - 0.9
            enemy['shoot_chance'] = 0.008
        elif mode in ['nebula', 'quantum']:
            enemy['drift'] = (SimpleRandom.random() * 2.8) - 1.4
            enemy['shoot_chance'] = 0.011
            enemy['speed'] += 0.6
        elif mode in ['ember', 'toxic', 'void']:
            enemy['drift'] = (SimpleRandom.random() * 1.6) - 0.8
            enemy['shoot_chance'] = 0.004
            enemy['speed'] += 1.0
        elif mode in ['frost', 'eclipse']:
            enemy['drift'] = (SimpleRandom.random() * 1.0) - 0.5
            enemy['shoot_chance'] = 0.009
        else:
            enemy['drift'] = (SimpleRandom.random() * 1.2) - 0.6

        # Base color used by enemy palettes.
        if mode == 'meteor':
            enemy['color'] = ORANGE
        elif mode == 'hybrid':
            enemy['color'] = CYAN
        elif mode == 'nebula':
            enemy['color'] = PURPLE
        elif mode == 'ember':
            enemy['color'] = (1, 0.4, 0.15, 1)
        elif mode == 'frost':
            enemy['color'] = (0.65, 0.9, 1, 1)
        elif mode == 'toxic':
            enemy['color'] = (0.45, 1, 0.35, 1)
        elif mode == 'quantum':
            enemy['color'] = (0.6, 0.5, 1, 1)
        elif mode == 'void':
            enemy['color'] = (0.35, 0.35, 0.45, 1)
        elif mode == 'eclipse':
            enemy['color'] = (0.95, 0.78, 0.2, 1)
        else:
            enemy['color'] = RED if variant == 'basic' else (YELLOW if variant == 'fast' else PURPLE)

        self.enemies.append(enemy)

    def get_boss_size(self, boss_kind):
        sizes = {
            'octopus': (84, 96),
            'grozilla': (96, 96),
            'growly': (90, 92),
            'manta': (96, 84),
            'scorpion': (96, 92),
            'wyvern': (98, 96),
            'serpent': (92, 96),
            'warden': (96, 98),
            'behemoth': (110, 100),
            'overseer': (108, 108),
            'tyrant': (130, 128),
        }
        return sizes.get(boss_kind, (84, 96))

    def spawn_stage_boss(self):
        """Spawn stage boss and its minions."""
        boss_kind = self.current_profile.get('boss_kind', 'octopus')
        boss_w, boss_h = self.get_boss_size(boss_kind)
        boss = {
            'x': SCREEN_WIDTH // 2,
            'y': SCREEN_HEIGHT - 120,
            'type': 'boss',
            'boss_kind': boss_kind,
            'width': boss_w,
            'height': boss_h,
            'health': 80 + (self.wave * 18),
            'max_health': 80 + (self.wave * 18),
            'speed': 2.4 + (self.wave * 0.08),
            'points': 700 + (self.wave * 40),
            'attack_timer': 0,
            'attack_mode': 0,
            'spawn_immunity': 1.5,
        }

        self.enemies.append(boss)
        self.start_boss_music()
        self.stage_phase = "boss"

        # Spawn minions alongside the boss.
        for idx in range(4):
            minion = {
                'x': boss['x'] - 120 + (idx * 80),
                'y': boss['y'] - 40 - (idx % 2) * 20,
                'width': 22,
                'height': 22,
                'type': 'minion',
                'variant': 'escort',
                'sprite_mode': self.current_profile.get('enemy_mode', 'alien'),
                'health': 1 + (self.wave // 4),
                'speed': 2.2 + (self.wave * 0.08),
                'points': 20 + (self.wave * 2),
                'color': ORANGE if boss_kind == 'grozilla' else (GREEN if boss_kind == 'growly' else PURPLE),
            }
            self.enemies.append(minion)

    def trigger_stage_clear(self, message="LEVEL Cleared"):
        """Start clear animation after boss defeat."""
        # Explode minions spawned with the boss.
        for enemy in self.enemies[:]:
            if enemy['type'] == 'minion':
                self.create_explosion(enemy['x'], enemy['y'], ORANGE)
                self.enemies.remove(enemy)

        self.enemy_bullets.clear()
        self.stage_phase = "clear_anim"
        self.clear_banner_timer = 2.0
        self.clear_message = message
        self.player_departing = True
        self.player['invulnerable'] = True
        self.start_background_music()

    def shoot(self):
        """Player shooting"""
        self.play_sound('shoot')
        if self.player['weapon_level'] == 1:
            # Single shot
            bullet = {
                'x': self.player['x'],
                'y': self.player['y'] + 20,
                'width': 4,
                'height': 12,
                'speed': 12,
                'color': YELLOW,
                'damage': 1
            }
            self.bullets.append(bullet)

        elif self.player['weapon_level'] == 2:
            # Double shot
            for offset in [-8, 8]:
                bullet = {
                    'x': self.player['x'] + offset,
                    'y': self.player['y'] + 20,
                    'width': 4,
                    'height': 12,
                    'speed': 12,
                    'color': CYAN,
                    'damage': 1
                }
                self.bullets.append(bullet)

        elif self.player['weapon_level'] >= 3:
            # Triple shot
            for offset, speed_x in [(-10, -1), (0, 0), (10, 1)]:
                bullet = {
                    'x': self.player['x'] + offset,
                    'y': self.player['y'] + 20,
                    'width': 4,
                    'height': 12,
                    'speed': 12,
                    'speed_x': speed_x,
                    'color': GREEN if speed_x == 0 else ORANGE,
                    'damage': 2 if speed_x == 0 else 1
                }
                self.bullets.append(bullet)

    def load_audio(self):
        """Load sound effects and music from sounds/."""
        self.sounds['shoot'] = (
            SoundLoader.load("sounds/shooting.wav.wav")
            or SoundLoader.load("sounds/shoot.wav")
        )
        self.bgm = SoundLoader.load("sounds/background.wav")
        self.boss_bgm = SoundLoader.load("sounds/boss_battle.wav")
        if self.bgm:
            self.bgm.loop = True
        if self.boss_bgm:
            self.boss_bgm.loop = True

    def play_sound(self, name):
        sound = self.sounds.get(name)
        if sound:
            sound.play()

    def _switch_music(self, music):
        if self.current_music is music:
            return
        if self.current_music:
            self.current_music.stop()
        self.current_music = music
        if self.current_music:
            self.current_music.play()

    def start_background_music(self):
        self._switch_music(self.bgm)

    def start_boss_music(self):
        self._switch_music(self.boss_bgm if self.boss_bgm else self.bgm)

    def create_explosion(self, x, y, color=ORANGE):
        """Create explosion particles"""
        for _ in range(10):
            self.particles.append(Particle(x, y, color))

    def update_ui(self):
        """Update UI elements"""
        self.health_label.text = f"Health: {self.player['lives']}"
        if self.ultra_boss_active:
            self.level_label.text = "Level: ULTRA"
        else:
            self.level_label.text = f"Level: {self.wave}"
        self.kills_label.text = f"Kills: {self.enemies_killed}"
        self.score_label.text = f"Score: {self.score}"

    def update(self, dt):
        """Main update loop"""
        if self.game_state == "playing":
            if self.stage_phase == "arrival":
                self.arrival_timer = max(0, self.arrival_timer - dt)
                self.player['y'] = min(100, self.player['y'] + 6.5)
                if self.arrival_timer <= 0:
                    if self.pending_ultra_spawn:
                        self.spawn_ultra_boss_trio()
                    else:
                        self.stage_phase = "enemies"

            elif self.stage_phase == "clear_anim":
                self.clear_banner_timer = max(0, self.clear_banner_timer - dt)
                if self.player_departing:
                    self.player['y'] += 8 + (self.wave * 0.15)
                if self.player['y'] > SCREEN_HEIGHT + 120:
                    self.player_departing = False
                    if self.ultra_boss_active:
                        self.game_state = "victory"
                    elif self.wave >= self.max_stages:
                        self.start_ultra_boss_encounter()
                    else:
                        self.player['weapon_level'] = min(4, self.player['weapon_level'] + 1)
                        self.setup_stage(self.wave + 1)
            else:
                # Player movement (left/right/up/down controls).
                if 'left' in self.keys_pressed:
                    self.player['x'] = max(25, self.player['x'] - self.player['speed'])
                if 'right' in self.keys_pressed:
                    self.player['x'] = min(SCREEN_WIDTH - 25, self.player['x'] + self.player['speed'])
                if 'up' in self.keys_pressed:
                    self.player['y'] = min(SCREEN_HEIGHT - 50, self.player['y'] + self.player['speed'])
                if 'down' in self.keys_pressed:
                    self.player['y'] = max(50, self.player['y'] - self.player['speed'])

                # Auto shooting.
                if self.shot_timer > 0:
                    self.shot_timer -= dt
                delay = 0.1 if self.rapid_fire else max(0.1, 0.26 - self.player['weapon_level'] * 0.025)
                if self.shot_timer <= 0:
                    self.shoot()
                    self.shot_timer = delay

                # Timers
                if self.player['invulnerable']:
                    self.player['invulnerable_timer'] -= 1
                    if self.player['invulnerable_timer'] <= 0:
                        self.player['invulnerable'] = False
                if self.shield_active:
                    self.shield_timer -= dt
                    if self.shield_timer <= 0:
                        self.shield_active = False
                if self.rapid_fire:
                    self.rapid_timer -= dt
                    if self.rapid_timer <= 0:
                        self.rapid_fire = False

                # Enemy spawn phase -> boss phase transition.
                if self.stage_phase == "enemies":
                    regular_enemies = [e for e in self.enemies if e['type'] == 'enemy']
                    target_on_screen = min(6 + self.wave, 14)

                    if self.stage_enemy_kills < self.stage_enemy_target:
                        self.spawn_timer -= dt
                        if len(regular_enemies) < target_on_screen and self.spawn_timer <= 0:
                            self.spawn_stage_enemy()
                            self.spawn_timer = max(0.14, 0.55 - self.wave * 0.02)
                    elif len(regular_enemies) == 0:
                        self.spawn_stage_boss()

                # Update enemies.
                for enemy in self.enemies[:]:
                    enemy_type = enemy['type']

                    if enemy_type == 'boss':
                        # Boss movement.
                        enemy['x'] += enemy['speed']
                        if enemy['x'] <= 70:
                            enemy['x'] = 70
                            enemy['speed'] = abs(enemy['speed'])
                        elif enemy['x'] >= SCREEN_WIDTH - 70:
                            enemy['x'] = SCREEN_WIDTH - 70
                            enemy['speed'] = -abs(enemy['speed'])

                        if enemy.get('spawn_immunity', 0) > 0:
                            enemy['spawn_immunity'] -= dt

                        enemy['attack_timer'] += dt
                        attack_delay = max(0.55, 1.1 - self.wave * 0.03)
                        if enemy['attack_timer'] >= attack_delay:
                            enemy['attack_timer'] = 0
                            boss_kind = enemy.get('boss_kind', 'octopus')

                            if boss_kind == 'octopus':
                                for vx in [-3, -1.5, 0, 1.5, 3]:
                                    self.enemy_bullets.append({
                                        'x': enemy['x'],
                                        'y': enemy['y'] - 20,
                                        'width': 6,
                                        'height': 12,
                                        'speed': -6,
                                        'speed_x': vx,
                                        'color': RED,
                                        'damage': 1,
                                        'kind': 'normal',
                                    })
                            elif boss_kind == 'grozilla':
                                if enemy['attack_mode'] == 0:
                                    # Pattern 1: throws meteors.
                                    for vx in [-2.8, -1.4, 0, 1.4, 2.8]:
                                        self.enemy_bullets.append({
                                            'x': enemy['x'],
                                            'y': enemy['y'] - 28,
                                            'width': 12,
                                            'height': 12,
                                            'speed': -5.8,
                                            'speed_x': vx,
                                            'color': ORANGE,
                                            'damage': 1,
                                            'kind': 'meteor_throw',
                                        })
                                else:
                                    # Pattern 2: roaring sound wave.
                                    for vx in [-4.5, -3, -1.5, 0, 1.5, 3, 4.5]:
                                        self.enemy_bullets.append({
                                            'x': enemy['x'],
                                            'y': enemy['y'] - 16,
                                            'width': 8,
                                            'height': 8,
                                            'speed': -4.2,
                                            'speed_x': vx,
                                            'growth': 0.3,
                                            'life': 1.5,
                                            'color': CYAN,
                                            'damage': 1,
                                            'kind': 'sound_wave',
                                        })
                                enemy['attack_mode'] = (enemy['attack_mode'] + 1) % 2
                            elif boss_kind == 'growly':
                                # Growly: boomerang knives from both hands.
                                for hand in [-1, 1]:
                                    self.enemy_bullets.append({
                                        'x': enemy['x'] + hand * 22,
                                        'y': enemy['y'] - 18,
                                        'width': 10,
                                        'height': 10,
                                        'speed': -6.8,
                                        'speed_x': hand * 6.2,
                                        'color': WHITE,
                                        'damage': 1,
                                        'kind': 'boomerang',
                                        'phase': 'out',
                                        'timer': 0,
                                        'out_time': 1.25,
                                        'owner': enemy,
                                    })
                            elif boss_kind in ['manta', 'warden']:
                                for vx in [-4.5, -3, -1.5, 0, 1.5, 3, 4.5]:
                                    self.enemy_bullets.append({
                                        'x': enemy['x'],
                                        'y': enemy['y'] - 20,
                                        'width': 7,
                                        'height': 10,
                                        'speed': -5.8,
                                        'speed_x': vx,
                                        'color': CYAN,
                                        'damage': 1,
                                        'kind': 'normal',
                                    })
                            elif boss_kind in ['scorpion', 'serpent', 'behemoth']:
                                for _ in range(3):
                                    vx = (SimpleRandom.random() * 6.0) - 3.0
                                    self.enemy_bullets.append({
                                        'x': enemy['x'] + vx * 4,
                                        'y': enemy['y'] - 26,
                                        'width': 10,
                                        'height': 10,
                                        'speed': -6.2,
                                        'speed_x': vx,
                                        'color': ORANGE,
                                        'damage': 1,
                                        'kind': 'meteor_throw',
                                    })
                            elif boss_kind in ['wyvern', 'overseer']:
                                for hand in [-1, 1]:
                                    self.enemy_bullets.append({
                                        'x': enemy['x'] + hand * 26,
                                        'y': enemy['y'] - 18,
                                        'width': 10,
                                        'height': 10,
                                        'speed': -3.5,
                                        'speed_x': hand * 4.6,
                                        'color': WHITE,
                                        'damage': 1,
                                        'kind': 'boomerang',
                                        'phase': 'out',
                                        'timer': 0,
                                        'out_time': 0.85,
                                        'owner': enemy,
                                    })
                            elif boss_kind == 'tyrant':
                                # Tyrant rotates through all major patterns.
                                mode = enemy['attack_mode'] % 3
                                if mode == 0:
                                    for vx in [-5, -3, -1, 1, 3, 5]:
                                        self.enemy_bullets.append({
                                            'x': enemy['x'],
                                            'y': enemy['y'] - 24,
                                            'width': 9,
                                            'height': 12,
                                            'speed': -6.2,
                                            'speed_x': vx,
                                            'color': RED,
                                            'damage': 1,
                                            'kind': 'normal',
                                        })
                                elif mode == 1:
                                    for vx in [-4.8, -3.2, -1.6, 0, 1.6, 3.2, 4.8]:
                                        self.enemy_bullets.append({
                                            'x': enemy['x'],
                                            'y': enemy['y'] - 18,
                                            'width': 10,
                                            'height': 10,
                                            'speed': -4.5,
                                            'speed_x': vx,
                                            'growth': 0.45,
                                            'life': 1.7,
                                            'color': CYAN,
                                            'damage': 1,
                                            'kind': 'sound_wave',
                                        })
                                else:
                                    for hand in [-1, 1]:
                                        self.enemy_bullets.append({
                                            'x': enemy['x'] + hand * 34,
                                            'y': enemy['y'] - 18,
                                            'width': 12,
                                            'height': 12,
                                            'speed': -4.2,
                                            'speed_x': hand * 6.4,
                                            'color': WHITE,
                                            'damage': 1,
                                            'kind': 'boomerang',
                                            'phase': 'out',
                                            'timer': 0,
                                            'out_time': 1.05,
                                            'owner': enemy,
                                        })
                                enemy['attack_mode'] = (enemy['attack_mode'] + 1) % 3

                    elif enemy_type == 'minion':
                        # Dive toward player during boss fight.
                        dx = self.player['x'] - enemy['x']
                        if dx > enemy['speed']:
                            enemy['x'] += enemy['speed']
                        elif dx < -enemy['speed']:
                            enemy['x'] -= enemy['speed']
                        enemy['y'] -= enemy['speed'] * 1.2
                        if SimpleRandom.random() < 0.01:
                            self.enemy_bullets.append({
                                'x': enemy['x'],
                                'y': enemy['y'] - 8,
                                'width': 5,
                                'height': 10,
                                'speed': -4.5,
                                'speed_x': 0,
                                'color': ORANGE,
                                'damage': 1,
                                'kind': 'normal',
                            })
                        if enemy['y'] < -80:
                            self.enemies.remove(enemy)

                    else:
                        # Stage regular enemies.
                        enemy['x'] += enemy.get('drift', 0)
                        enemy['y'] -= enemy['speed']
                        if enemy['sprite_mode'] != 'meteor' and SimpleRandom.random() < enemy.get('shoot_chance', 0):
                            self.enemy_bullets.append({
                                'x': enemy['x'],
                                'y': enemy['y'] - 10,
                                'width': 4,
                                'height': 8,
                                'speed': -5,
                                'speed_x': 0,
                                'color': RED,
                                'damage': 1,
                                'kind': 'normal',
                            })
                        if enemy['y'] < -80 or enemy['x'] < -80 or enemy['x'] > SCREEN_WIDTH + 80:
                            self.enemies.remove(enemy)

                # Update player bullets.
                for bullet in self.bullets[:]:
                    bullet['y'] += bullet['speed']
                    if 'speed_x' in bullet:
                        bullet['x'] += bullet['speed_x']
                    if bullet['y'] > SCREEN_HEIGHT + 40 or bullet['x'] < -40 or bullet['x'] > SCREEN_WIDTH + 40:
                        self.bullets.remove(bullet)

                # Update enemy bullets (supports boomerangs and sound waves).
                for bullet in self.enemy_bullets[:]:
                    kind = bullet.get('kind', 'normal')
                    if kind == 'boomerang':
                        bullet['timer'] += dt
                        if bullet.get('phase') == 'out':
                            bullet['x'] += bullet['speed_x']
                            bullet['y'] += bullet['speed']
                            if bullet['timer'] >= bullet.get('out_time', 0.55):
                                bullet['phase'] = 'back'
                        else:
                            owner = bullet.get('owner')
                            target_x = owner['x'] if owner in self.enemies else SCREEN_WIDTH // 2
                            target_y = owner['y'] if owner in self.enemies else SCREEN_HEIGHT - 80
                            dx = target_x - bullet['x']
                            dy = target_y - bullet['y']
                            bullet['x'] += dx * 0.22
                            bullet['y'] += dy * 0.22
                            if abs(dx) < 8 and abs(dy) < 8:
                                self.enemy_bullets.remove(bullet)
                                continue
                    else:
                        bullet['y'] += bullet['speed']
                        bullet['x'] += bullet.get('speed_x', 0)
                        if kind == 'sound_wave':
                            bullet['width'] = min(24, bullet['width'] + bullet.get('growth', 0))
                            bullet['height'] = min(24, bullet['height'] + bullet.get('growth', 0))
                            bullet['life'] = bullet.get('life', 0) - dt
                            if bullet['life'] <= 0:
                                self.enemy_bullets.remove(bullet)
                                continue

                    if (bullet['y'] < -60 or bullet['y'] > SCREEN_HEIGHT + 60 or
                            bullet['x'] < -80 or bullet['x'] > SCREEN_WIDTH + 80):
                        self.enemy_bullets.remove(bullet)

                # Power-ups drift down slowly.
                for powerup in self.powerups[:]:
                    powerup['y'] -= powerup.get('fall_speed', 1.0)
                    powerup['x'] += powerup.get('drift', 0.0)
                    if powerup['y'] < -30:
                        self.powerups.remove(powerup)

                # Collisions.
                player_rect = {
                    'x': self.player['x'] - 15,
                    'y': self.player['y'] - 15,
                    'width': 30,
                    'height': 30
                }

                for bullet in self.bullets[:]:
                    for enemy in self.enemies[:]:
                        if (bullet['x'] > enemy['x'] - enemy['width'] // 2 and
                                bullet['x'] < enemy['x'] + enemy['width'] // 2 and
                                bullet['y'] > enemy['y'] - enemy['height'] // 2 and
                                bullet['y'] < enemy['y'] + enemy['height'] // 2):

                            if bullet in self.bullets:
                                self.bullets.remove(bullet)

                            if enemy['type'] == 'boss' and enemy.get('spawn_immunity', 0) > 0:
                                break

                            enemy['health'] -= bullet['damage']
                            if enemy['health'] <= 0:
                                self.score += enemy['points']
                                self.enemies_killed += 1
                                self.create_explosion(enemy['x'], enemy['y'])
                                self.play_sound('explosion')

                                if enemy in self.enemies:
                                    self.enemies.remove(enemy)

                                if enemy['type'] == 'enemy':
                                    self.stage_enemy_kills += 1
                                    if SimpleRandom.random() < POWERUP_DROP_CHANCE:
                                        self.powerups.append({
                                            'x': enemy['x'],
                                            'y': enemy['y'],
                                            'type': SimpleRandom.choice(['health', 'weapon', 'shield', 'rapid']),
                                            'width': 20,
                                            'height': 20,
                                            'fall_speed': 0.8 + (SimpleRandom.random() * 0.7),
                                            'drift': (SimpleRandom.random() * 0.5) - 0.25,
                                        })
                                elif enemy['type'] == 'boss':
                                    if self.stage_phase == "ultra_boss":
                                        remaining_bosses = [e for e in self.enemies if e.get('type') == 'boss']
                                        if len(remaining_bosses) == 0:
                                            self.trigger_stage_clear("The Tyrant Has Fallen")
                                    else:
                                        self.trigger_stage_clear()
                            break

                if not self.player['invulnerable'] and not self.shield_active:
                    for bullet in self.enemy_bullets[:]:
                        if (bullet['x'] > player_rect['x'] and
                                bullet['x'] < player_rect['x'] + player_rect['width'] and
                                bullet['y'] > player_rect['y'] and
                                bullet['y'] < player_rect['y'] + player_rect['height']):
                            self.enemy_bullets.remove(bullet)
                            self.player['lives'] -= 1
                            self.player['invulnerable'] = True
                            self.player['invulnerable_timer'] = 120
                            self.create_explosion(self.player['x'], self.player['y'], RED)
                            if self.player['lives'] <= 0:
                                self.game_state = "game_over"
                                if self.current_music:
                                    self.current_music.stop()
                            break

                if not self.player['invulnerable'] and not self.shield_active:
                    for enemy in self.enemies[:]:
                        if (self.player['x'] - 15 < enemy['x'] + enemy['width'] // 2 and
                                self.player['x'] + 15 > enemy['x'] - enemy['width'] // 2 and
                                self.player['y'] - 15 < enemy['y'] + enemy['height'] // 2 and
                                self.player['y'] + 15 > enemy['y'] - enemy['height'] // 2):
                            if enemy['type'] != 'boss' and enemy in self.enemies:
                                self.enemies.remove(enemy)
                            self.player['lives'] -= 1
                            self.player['invulnerable'] = True
                            self.player['invulnerable_timer'] = 120
                            self.create_explosion(self.player['x'], self.player['y'], RED)
                            if self.player['lives'] <= 0:
                                self.game_state = "game_over"
                            break

                # Power-up collection.
                for powerup in self.powerups[:]:
                    if (self.player['x'] - 15 < powerup['x'] + 10 and
                            self.player['x'] + 15 > powerup['x'] - 10 and
                            self.player['y'] - 15 < powerup['y'] + 10 and
                            self.player['y'] + 15 > powerup['y'] - 10):
                        self.powerups.remove(powerup)
                        if powerup['type'] == 'health':
                            self.player['lives'] = min(5, self.player['lives'] + 1)
                        elif powerup['type'] == 'weapon':
                            self.player['weapon_level'] = min(4, self.player['weapon_level'] + 1)
                        elif powerup['type'] == 'shield':
                            self.shield_active = True
                            self.shield_timer = 5
                        elif powerup['type'] == 'rapid':
                            self.rapid_fire = True
                            self.rapid_timer = 5

            self.update_ui()

        # Update starfield
        self.starfield.update()
        self.particles = [p for p in self.particles if p.update()]

        # Redraw
        self.canvas.clear()
        self.draw()

    def draw(self):
        """Draw everything"""
        with self.canvas:
            # Fill full window (letterbox region)
            Color(0, 0, 0, 1)
            Rectangle(pos=(0, 0), size=self.size)

            # Render game in virtual resolution with responsive scaling.
            scale = min(self.width / SCREEN_WIDTH, self.height / SCREEN_HEIGHT)
            offset_x = (self.width - (SCREEN_WIDTH * scale)) / 2
            offset_y = (self.height - (SCREEN_HEIGHT * scale)) / 2

            PushMatrix()
            Translate(offset_x, offset_y, 0)
            Scale(scale, scale, 1)

            self.draw_stage_background()

            if self.game_state == "menu":
                self.draw_menu()
            elif self.game_state == "playing":
                self.draw_game()
            elif self.game_state == "game_over":
                self.draw_game_over()
            elif self.game_state == "victory":
                self.draw_victory()

            PopMatrix()

    def draw_text(self, text, center_x, center_y, color=WHITE, font_size=26):
        """Draw centered text with Kivy core text."""
        core_label = CoreLabel(text=text, font_size=font_size, color=color)
        core_label.refresh()
        texture = core_label.texture
        Color(*color)
        Rectangle(
            texture=texture,
            pos=(center_x - texture.width / 2, center_y - texture.height / 2),
            size=texture.size
        )

    def draw_text_left(self, text, x, y, color=WHITE, font_size=22):
        """Draw left-aligned text."""
        core_label = CoreLabel(text=text, font_size=font_size, color=color)
        core_label.refresh()
        texture = core_label.texture
        Color(*color)
        Rectangle(texture=texture, pos=(x, y), size=texture.size)

    def px(self, value):
        return int(value // PIXEL_SIZE) * PIXEL_SIZE

    def draw_pixel_sprite(self, center_x, center_y, pattern, palette, scale=PIXEL_SIZE):
        """Draw pixel-art sprite from text pattern."""
        height = len(pattern)
        width = max(len(row) for row in pattern)
        origin_x = self.px(center_x - (width * scale) / 2)
        origin_y = self.px(center_y - (height * scale) / 2)

        for row_idx, row in enumerate(pattern):
            row = row.ljust(width, ".")
            for col_idx, pixel in enumerate(row):
                color = palette.get(pixel)
                if color:
                    Color(*color)
                    x = origin_x + (col_idx * scale)
                    y = origin_y + ((height - 1 - row_idx) * scale)

    def darken(self, color, amount=0.35):
        r, g, b, a = color
        return (max(0, r - amount), max(0, g - amount), max(0, b - amount), a)

                self.draw_pixel_sprite(self.player['x'], self.player['y'], ship_pattern, ship_palette, scale=2)

                # Draw engine fire if moving
                if 'up' in self.keys_pressed:
                    Color(1, 0.5, 0, 1)
                    Rectangle(
                        pos=(self.px(self.player['x'] - 8), self.px(self.player['y'] - 22)),
                        size=(16, 12)
                    )

            # Left-side HUD (always visible in-game)
            self.draw_text_left(f"Health: {self.player['lives']}", 20, SCREEN_HEIGHT - 44, color=(1, 0.35, 0.35, 1), font_size=22)
            self.draw_text_left(f"Level: {self.wave}", 20, SCREEN_HEIGHT - 74, color=WHITE, font_size=20)
            self.draw_text_left(f"Kills: {self.enemies_killed}", 20, SCREEN_HEIGHT - 104, color=(0.7, 1, 0.7, 1), font_size=20)
            self.draw_text_left(f"Score: {self.score}", 20, SCREEN_HEIGHT - 134, color=WHITE, font_size=18)

            if self.stage_phase == "arrival":
                self.draw_text(self.arrival_text, SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2, color=CYAN, font_size=40)
            elif self.stage_phase == "clear_anim":
                self.draw_text(self.clear_message, SCREEN_WIDTH // 2, SCREEN_HEIGHT // 2, color=GREEN, font_size=44)

    def draw_game_over(self):
        """Draw game over screen"""
        with self.canvas:
            self.draw_text("GAME OVER", SCREEN_WIDTH // 2, 520, color=RED, font_size=56)
            self.draw_text(f"Final Score: {self.score}", SCREEN_WIDTH // 2, 400, color=WHITE, font_size=30)
            self.draw_text(f"Level Reached: {self.wave}", SCREEN_WIDTH // 2, 350, color=WHITE, font_size=24)
            self.draw_text("Press R to Restart or Q to Quit", SCREEN_WIDTH // 2, 260, color=WHITE, font_size=22)

    def draw_victory(self):
        """Draw victory screen"""
        with self.canvas:
            self.draw_text("VICTORY!", SCREEN_WIDTH // 2, 540, color=GREEN, font_size=58)
            self.draw_text("You cleared all 10 stages", SCREEN_WIDTH // 2, 430, color=WHITE, font_size=28)
            self.draw_text(f"Final Score: {self.score}", SCREEN_WIDTH // 2, 370, color=WHITE, font_size=28)
            self.draw_text("Press R to Play Again or Q to Quit", SCREEN_WIDTH // 2, 270, color=WHITE, font_size=22)
